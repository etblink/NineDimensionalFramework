# Define sensitivity analysis function
def compute_sensitivity(param_values, energy_values):
    # Compute percentage change in energy contribution per unit change in parameter
    sensitivity = np.gradient(energy_values, param_values)
    normalized_sensitivity = sensitivity / np.max(np.abs(sensitivity))  # Normalize to max sensitivity
    return sensitivity, normalized_sensitivity
